3. BERT Sentence Embeddings & Sentence Similarity

We use BERT's [CLS] token output or mean pooling as the sentence embedding.

Source cell 12
@torch.no_grad()
def get_bert_embeddings(texts, pooling='mean'):
    """
    Extract sentence embeddings with BERT.

    Args:
        pooling: 'cls' = [CLS] token, 'mean' = average over all tokens
    """
    encoded = bert_tokenizer(
        texts, padding=True, truncation=True,
        max_length=128, return_tensors='pt'
    ).to(device)

    outputs = bert_model(**encoded)

    if pooling == 'cls':
        # Output of the [CLS] token
        embeddings = outputs.last_hidden_state[:, 0, :]
    else:
        # Mean pooling (respecting the attention mask)
        attention_mask = encoded['attention_mask'].unsqueeze(-1)
        token_embeds = outputs.last_hidden_state
        embeddings = (token_embeds * attention_mask).sum(1) / attention_mask.sum(1)

    # L2 normalize
    embeddings = F.normalize(embeddings, p=2, dim=1)
    return embeddings.cpu().numpy()

4. Masked Language Modeling (MLM)

BERT's core training objective: hide part of a sentence with [MASK] and learn to predict the original word.

Input:  "The [MASK] is playing in the park."
Predict: dog (95%), cat (2%), child (1%), ...

Through this, BERT learns to understand context bidirectionally.

Source cell 15
@torch.no_grad()
def predict_masked(text, top_k=10):
    """
    Predict the word that fits the position of the [MASK] token.
    """
    inputs = bert_tokenizer(text, return_tensors='pt').to(device)
    outputs = bert_mlm(**inputs)
    logits = outputs.logits  # (1, seq_len, vocab_size)

    # Find the [MASK] positions
    mask_idx = (inputs['input_ids'][0] == bert_tokenizer.mask_token_id).nonzero(as_tuple=True)[0]

    results = []
    for idx in mask_idx:
        mask_logits = logits[0, idx]
        probs = torch.softmax(mask_logits, dim=0)
        top_probs, top_indices = probs.topk(top_k)

        predictions = []
        for prob, token_id in zip(top_probs, top_indices):
            word = bert_tokenizer.decode(token_id.item())
            predictions.append((word, prob.item()))
        results.append(predictions)

    return results

  • Contrastive learning over 400M (image, text) pairs

                                                ├── Cosine Similarity
Text  ──→ [Text Encoder] ──→ Text Embedding  ─┘

7. CLIP Zero-Shot Classification

CLIP's most impressive ability: classifying images without training.

Principle: compare the cosine similarity between the image embedding and each class-text embedding, and pick the class of the most similar text.

Source cell 22
@torch.no_grad()
def clip_zero_shot_classify(images, class_texts):
    """
    Perform zero-shot classification with CLIP.

    Args:
        images: list of PIL Image
        class_texts: list of str (e.g., ["a photo of a dog", "a photo of a cat"])

    Returns:
        probs: (N, C) probability matrix
        pred_indices: (N,) predicted class indices
    """
    inputs = clip_processor(
        text=class_texts, images=images,
        return_tensors="pt", padding=True
    ).to(device)

    outputs = clip_model(**inputs)
    logits = outputs.logits_per_image  # (N, C)
    probs = logits.softmax(dim=1).cpu().numpy()
    pred_indices = probs.argmax(axis=1)

    return probs, pred_indices

@torch.no_grad()
def visualize_next_token(text, top_k=10):
    """Visualize the probability distribution of the next token."""
    inputs = llm_tokenizer(text, return_tensors='pt').to(device)
    outputs = llm_model(**inputs)
    
    # Logits at the last position → next-token probabilities
    next_logits = outputs.logits[0, -1, :]
    probs = torch.softmax(next_logits, dim=0)
    top_probs, top_ids = probs.topk(top_k)
    
    tokens = [llm_tokenizer.decode(idx.item()) for idx in top_ids]
    probs_list = top_probs.cpu().numpy()
    
    return tokens, probs_list

3. Temperature & Sampling Strategies

There are several ways an LLM can pick the next token:

Strategy Description
Greedy Always pick the most probable token (repetitive)
Temperature Smooth/sharpen the probability distribution
Top-k Sample only from the top-k candidates
Top-p (nucleus) Sample from candidates up to cumulative probability p
Source cell 10
def generate_text(prompt, max_new_tokens=60, temperature=1.0, top_k=50, top_p=0.9, do_sample=True):
    """Generate text with GPT-2."""
    inputs = llm_tokenizer(prompt, return_tensors='pt').to(device)
    
    with torch.no_grad():
        outputs = llm_model.generate(
            **inputs,
            max_new_tokens=max_new_tokens,
            temperature=temperature,
            top_k=top_k,
            top_p=top_p,
            do_sample=do_sample,
            pad_token_id=llm_tokenizer.eos_token_id,
        )
    
    generated = llm_tokenizer.decode(outputs[0], skip_special_tokens=True)
    return generated

from qwen_vl_utils import process_vision_info

def ask_vlm(image_path, question, max_new_tokens=256, temperature=0.7):
    """
    Send an image and a question to Qwen3-VL and get the response.
    
    Args:
        image_path: str, path to the image file
        question: str, the question text
    Returns:
        str: model response
    """
    messages = [
        {
            "role": "user",
            "content": [
                {"type": "image", "image": f"file://{os.path.abspath(image_path)}"},
                {"type": "text", "text": question},
            ],
        }
    ]
    
    text = vlm_processor.apply_chat_template(messages, tokenize=False, add_generation_prompt=True)
    image_inputs, video_inputs = process_vision_info(messages)
    
    inputs = vlm_processor(
        text=[text],
        images=image_inputs,
        videos=video_inputs,
        padding=True,
        return_tensors="pt",
    ).to(vlm_model.device)
    
    with torch.no_grad():
        generated_ids = vlm_model.generate(
            **inputs,
            max_new_tokens=max_new_tokens,
            temperature=temperature,
            do_sample=True if temperature > 0 else False,
        )
    
    # Strip the input portion
    generated_ids_trimmed = [
        out_ids[len(in_ids):]
        for in_ids, out_ids in zip(inputs.input_ids, generated_ids)
    ]
    
    response = vlm_processor.batch_decode(
        generated_ids_trimmed, skip_special_tokens=True,
        clean_up_tokenization_spaces=False
    )[0]
    
    return response.strip()

def multi_turn_conversation(image_path, conversation_questions):
    """
    Run a multi-turn conversation about an image.
    """
    messages = []
    results = []
    
    for i, question in enumerate(conversation_questions):
        if i == 0:
            # Include the image only in the first question
            messages.append({
                "role": "user",
                "content": [
                    {"type": "image", "image": f"file://{os.path.abspath(image_path)}"},
                    {"type": "text", "text": question},
                ],
            })
        else:
            messages.append({
                "role": "user",
                "content": [{"type": "text", "text": question}],
            })
        
        text = vlm_processor.apply_chat_template(messages, tokenize=False, add_generation_prompt=True)
        image_inputs, video_inputs = process_vision_info(messages)
        inputs = vlm_processor(
            text=[text], images=image_inputs, videos=video_inputs,
            padding=True, return_tensors="pt"
        ).to(vlm_model.device)
        
        with torch.no_grad():
            gen_ids = vlm_model.generate(**inputs, max_new_tokens=200)
        
        trimmed = [o[len(i):] for i, o in zip(inputs.input_ids, gen_ids)]
        response = vlm_processor.batch_decode(trimmed, skip_special_tokens=True)[0].strip()
        
        # Add the response to the conversation history
        messages.append({"role": "assistant", "content": [{"type": "text", "text": response}]})
        results.append((question, response))
    
    return results
